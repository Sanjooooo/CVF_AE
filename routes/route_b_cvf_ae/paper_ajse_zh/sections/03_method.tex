\section{CVF-AE 方法}
\label{sec:method}

本文提出的 CVF-AE 在自适应进化搜索中引入约束可行性场（Constraint Viability Field, CVF）。CVF 不改写目标函数，也不等同于评价后修复，而是在候选生成阶段由障碍物、禁飞区、高度、转弯、边界和风险信息给出有界方向增量，使控制点在评价前向更可行、更稳健的区域移动。

\subsection{总体框架}

CVF-AE 的一次迭代包括基础进化搜索、状态识别、CVF 候选生成、稀疏可行性保持和可行性优先接收。本文的稀疏可行性保持包括少量候选的局部修复后处理，用于维持可行性但不替代 CVF 候选方向。给定种群 $\{\mathbf{X}_i\}_{i=1}^{N}$，算法先按式 \eqref{eq:objective} 和式 \eqref{eq:total-violation} 评价个体并记录最优可行解、可行率、平均违反量和近期改进；基础算子生成 $\mathbf{D}_{\mathrm{AE}}$，被触发的个体再构造 $\mathbf{D}_{\mathrm{CVF}}$ 与 $\mathbf{D}_{\mathrm{Q}}$。

候选生成可写为
\begin{equation}
    \mathbf{X}_{\mathrm{AE}} =
    \Pi_{\Omega}
    \left(
    (1-\mu_t)(\mathbf{X}_i+\mathbf{D}_{\mathrm{AE}})
    +\mu_t\mathbf{X}_{\mathrm{elite}}
    \right),
    \label{eq:ae-candidate}
\end{equation}
\begin{equation}
    \begin{aligned}
    \mathbf{D}_s
    &=
    \alpha_s \mathbf{D}_{\mathrm{AE}}
    + \beta_s \mathbf{D}_{\mathrm{CVF}}\\
    &\quad
    + \gamma_s \mathbf{D}_{\mathrm{Q}},\\
    \mathbf{X}_{\mathrm{CVF}}
    &=
    \Pi_{\Omega}
    \Bigl(
    (1-\mu_t)(\mathbf{X}_i+\mathbf{D}_s)\\
    &\quad
    +\mu_t\mathbf{X}_{\mathrm{elite}}
    \Bigr),
    \end{aligned}
    \label{eq:cvf-candidate}
\end{equation}
其中 $\mathbf{D}_s$ 为状态融合方向，$\Pi_{\Omega}(\cdot)$ 为边界投影，$\mathbf{X}_{\mathrm{elite}}$ 为精英池采样个体，$\mu_t$ 为小权重锚定系数，$\alpha_s,\beta_s,\gamma_s$ 由状态 $s$ 决定。CVF 分支只为被稀疏触发的个体提供额外候选，并需通过局部接收规则；稀疏可行性保持则作为候选后处理独立执行。

基础 AE 方向 $\mathbf D_{\mathrm{AE}}$ 按状态选择四类搜索模式。令 $\tau=t/T$，$\mathbf d_{\mathrm{best}}=\mathbf X^*-\mathbf X_i$，$\mathbf d_{\mathrm{elite}}=\bar{\mathbf X}_{\mathrm{elite}}-\mathbf X_i$，$\mathbf d_{\mathrm{sample}}=\mathbf X_{\mathrm{elite}}-\mathbf X_i$，$\mathbf d_{\mathrm{ref}}=\mathbf X_{\mathrm{ref}}-\mathbf X_i$；$\mathbf d_{ab}$ 和 $\mathbf d_{cd}$ 为随机差分向量，$\boldsymbol\xi$ 和 $\boldsymbol\zeta$ 分别为范围缩放均匀扰动和高斯扰动。基础方向统一写为
\begin{equation}
    \begin{aligned}
    \mathbf D_{\mathrm{AE}}
    ={}&c_b\mathbf r_b\odot\mathbf d_{\mathrm{best}}
    +c_e\mathbf r_e\odot\mathbf d_{\mathrm{elite}}\\
    &+c_s\mathbf r_s\odot\mathbf d_{\mathrm{sample}}
    +c_r\mathbf r_r\odot\mathbf d_{\mathrm{ref}}\\
    &+c_1\mathbf d_{ab}+c_2\mathbf d_{cd}
    +c_u\boldsymbol\xi+c_g\boldsymbol\zeta .
    \end{aligned}
\label{eq:ae-state-direction}
\end{equation}
其中各 $\mathbf r$ 为 $[0,1]^{3K}$ 上的独立随机向量，$\odot$ 表示逐维乘积。四个状态依次为可行性形成、可行性保持、质量细化和停滞恢复，非零系数见表 \ref{tab:base-ae-coefficients}。

\subsection{约束状态识别}

CVF-AE 由种群状态调节约束引导强度。令 $\rho_f$、$\bar{V}$、$D_{\mathrm{pop}}$ 和 $\Delta J_{\mathrm{recent}}$ 分别表示可行率、平均违反量、多样性和近期目标改进：
\begin{equation}
    \rho_f=\frac{1}{N}\sum_{i=1}^{N}\mathbb I(\mathbf X_i\ \mathrm{feasible}),
    \qquad
    \bar V=\frac{1}{N}\sum_{i=1}^{N}V(\mathbf X_i),
    \label{eq:state-feasibility}
\end{equation}
\begin{equation}
    D_{\mathrm{pop}}=
    \frac{1}{3K}\sum_{j=1}^{3K}
    \mathrm{std}\left(\frac{X_{1j},\ldots,X_{Nj}}{u_j-l_j}\right),
    \label{eq:state-diversity}
\end{equation}
\begin{equation}
    \Delta J_{\mathrm{recent}}=
    \frac{\max(0,J^*_{t-w}-J^*_{t})}{\max(1,|J^*_{t-w}|)} ,
    \qquad w=15 .
    \label{eq:recent-improvement}
\end{equation}
当 $\Delta J_{\mathrm{recent}}<10^{-4}$ 时认为搜索停滞。观测状态按以下互斥顺序判定：
\begin{equation}
s_{\mathrm{obs}}=
\begin{cases}
4, & \begin{aligned}[t]
        &\Delta J_{\mathrm{recent}}<10^{-4}\\[-1mm]
        &{}\land (D_{\mathrm{pop}}<0.08\lor t>2w),
    \end{aligned}\\
1, & \rho_f<0.20\lor \bar V>10,\\
2, & \rho_f<0.65,\\
3, & \mathrm{otherwise}.
\end{cases}
\label{eq:observed-state}
\end{equation}
式 \eqref{eq:observed-state} 为观测模板。正式运行采用保守调度：默认保持状态；当当前最优解尚不可行且 $(\rho_f\leq0.10\lor \bar V\geq18)$ 时请求形成状态；当搜索停滞、$\rho_f<0.55$ 且 $\bar V\geq6$ 时请求恢复状态。非保持状态需连续确认 $2$ 代才生效。质量细化状态保留为观测和权重模板；若不满足形成或恢复条件，正式运行回到保持状态，以减少状态跳变和过度干预。经保守调度后得到生效状态 $s_t$；下文所有 $\alpha_s,\beta_s,\gamma_s$ 和 CVF 权重均使用生效状态。

\subsection{约束可行性场构建}

CVF 并非目标函数或约束函数的解析梯度，而是由局部约束压力构造的有界启发式候选方向，仅用于候选生成阶段。CVF 在内部控制点空间中构造，并使用低密度路径采样集 $\widetilde{\mathcal P}=\{\widetilde{\mathbf p}_m\}_{m=1}^{\widetilde M}$ 降低成本。采样点压力先分配到最近内部控制点：
\begin{equation}
    k^*(m)=\arg\min_{1\leq k\leq K}\|\widetilde{\mathbf p}_m-\mathbf{q}_k\|_2 .
    \label{eq:nearest-control}
\end{equation}
对第 $k$ 个控制点，其 CVF 方向由多个场分量叠加得到：
\begin{equation}
    \begin{aligned}
    \mathbf{F}_k =
    \sum_{c\in\mathcal{C}}
    \eta_c(s)\mathbf{F}_{k,c},\\
    \mathcal{C}=
    \{\mathrm{obs},\mathrm{nfz},\mathrm{risk},\mathrm{alt},\mathrm{curv},\mathrm{bnd}\},
    \end{aligned}
    \label{eq:cvf-field}
\end{equation}
其中 $\mathbf{F}_{k,c}$ 为第 $c$ 类压力方向，$\eta_c(s)$ 为状态相关权重；局部违反程度通过各压力分量 $\mathbf{F}_{k,c}$ 的幅值体现。按 $(\mathrm{obs},\mathrm{nfz},\mathrm{risk},\mathrm{alt},\mathrm{curv},\mathrm{bnd})$ 排列，四类状态权重和融合权重为

\begin{tabular}{@{}c c c@{}}

$s$ & $\eta_s$ & $(\alpha_s,\beta_s,\gamma_s)$\\
\midrule
1 & $(1.20,1.15,0.85,1.10,0.85,0.65)$ & $(0.80,0.40,0.04)$\\
2 & $(0.95,0.95,0.85,0.90,0.75,0.55)$ & $(0.90,0.38,0.06)$\\
3 & $(0.35,0.35,0.55,0.40,0.95,0.35)$ & $(1.00,0.38,0.06)$\\

其中细化权重主要保留为完整状态模板。

若多个采样点映射到同一控制点，则累加压力并按命中次数平均。记 $n_k$ 为映射到第 $k$ 个控制点且压力非零的采样点数量，非曲率增量为
\begin{equation}
    \Delta \mathbf q_k^{\mathrm{path}}=
    \begin{cases}
    \dfrac{1}{n_k}\sum_{m:k^*(m)=k}\mathbf F(\widetilde{\mathbf p}_m), & n_k>0,\\[6pt]
    \mathbf 0, & n_k=0,
    \end{cases}
    \label{eq:cvf-average-pressure}
\end{equation}
其中 $\mathbf F(\widetilde{\mathbf p}_m)$ 为采样点处障碍物、禁飞区、风险、高度和边界压力的加权和。曲率分量在控制点序列上计算，并与 $\Delta \mathbf q_k^{\mathrm{path}}$ 相加得到 $\Delta\mathbf q_k$。

\paragraph{障碍物分量}
对于轴对齐障碍物 $\mathcal B=[x_1,x_2]\times[y_1,y_2]\times[z_1,z_2]$，令 $\mathbf c_{\mathcal B}(\mathbf p)$ 为 $\mathbf p$ 在该盒体上的最近点。若 $\mathbf p$ 位于障碍物外部，定义
\begin{equation}
    \mathbf d_o=
    [p_x-c_x,\ p_y-c_y,\ 0.35(p_z-c_z)]^\mathsf T,\qquad
    d_o=\|\mathbf d_o\|_2 .
    \label{eq:obstacle-distance-vector}
\end{equation}
当 $0<d_o\leq r_o$ 时，障碍物压力为
\begin{equation}
    \mathbf f_{\mathrm{obs}}(\mathbf p;\mathcal B)=
    \frac{r_o-d_o}{r_o}\frac{\mathbf d_o}{d_o}.
    \label{eq:obstacle-buffer-pressure}
\end{equation}
若 $\mathbf p$ 位于障碍物内部，则选择最近的水平外侧面法向 $\mathbf n_{\min}\in\{[-1,0,0]^\mathsf T,[1,0,0]^\mathsf T,[0,-1,0]^\mathsf T,[0,1,0]^\mathsf T\}$，并令
\begin{equation}
    \mathbf f_{\mathrm{obs}}(\mathbf p;\mathcal B)=
    (r_o+d_{\min})\mathbf n_{\min},
    \label{eq:obstacle-inside-pressure}
\end{equation}
其中 $d_{\min}$ 为到最近水平面的穿透深度，$r_o=7$。式 \eqref{eq:obstacle-distance-vector} 压缩垂向距离，以抑制单纯大幅爬升绕障。

\paragraph{禁飞区分量}
设柱状禁飞区为 $\mathcal Z=(\mathbf c_z,r_z,z_l,z_u)$。若 $p_z\notin [z_l,z_u]$，该分量为零；否则令 $\mathbf d_z=[p_x-c_{z,x},p_y-c_{z,y},0]^\mathsf T$，$d_z=\|\mathbf d_z\|_2$。禁飞区径向压力定义为
\begin{equation}
    \mathbf f_{\mathrm{nfz}}(\mathbf p;\mathcal Z)=
    \begin{cases}
    (r_z-d_z+r_n)\dfrac{\mathbf d_z}{d_z}, & d_z\leq r_z,\\[6pt]
    \dfrac{r_z+r_n-d_z}{r_n}\dfrac{\mathbf d_z}{d_z}, & r_z<d_z\leq r_z+r_n,\\[6pt]
    \mathbf 0, & d_z>r_z+r_n,
    \end{cases}
    \label{eq:nfz-pressure}
\end{equation}
其中 $r_n=7$ 为禁飞区缓冲影响半径；当 $d_z$ 接近零时使用固定水平单位方向避免除零。

\paragraph{风险分量}
风险分量不作为硬约束，而是质量压力。对风险热点 $\mathcal H_h=(\mathbf c_h,\sigma_h,a_h)$，令 $\widetilde{\mathbf d}_h=[p_x-c_{h,x},p_y-c_{h,y},0.35(p_z-c_{h,z})]^\mathsf T$，并定义
\begin{equation}
    \phi_h(\mathbf p)=
    a_h\exp\left(-\frac{\|\widetilde{\mathbf d}_h\|_2^2}{2\sigma_h^2}\right).
    \label{eq:risk-hotspot}
\end{equation}
仅当 $\phi_h(\mathbf p)>\phi_0$ 时触发风险压力：
\begin{equation}
    \mathbf f_{\mathrm{risk}}(\mathbf p;\mathcal H_h)=
    \kappa_r(\phi_h(\mathbf p)-\phi_0)
    \frac{\widetilde{\mathbf d}_h}{\|\widetilde{\mathbf d}_h\|_2+\epsilon_l},
    \label{eq:risk-pressure}
\end{equation}
其中 $\phi_0=0.24$，$\kappa_r=1.8$。该分量仅作为质量压力，不改变硬约束可行性定义。

\paragraph{高度和边界分量}
记 $z_l=z_{\min}$，$z_u=z_{\max}$。高度分量采用分段形式：
\begin{equation}
f_{\mathrm{alt},z}(\mathbf p)=
\begin{cases}
z_l-p_z, & p_z<z_l,\\
z_u-p_z, & p_z>z_u,\\
0.35(z_l+b_z-p_z), & z_l\leq p_z<z_l+b_z,\\
-0.35(p_z-z_u+b_z), & z_u-b_z<p_z\leq z_u,\\
0, & \mathrm{otherwise},
\end{cases}
\label{eq:altitude-pressure}
\end{equation}
并叠加弱参考高度项 $0.04(z_{\mathrm{ref}}-p_z)$，其中 $b_z=3$。边界分量只在水平安全带内生效：
\begin{equation}
\begin{aligned}
f_{\mathrm{bnd},x}(\mathbf p)&=
\begin{cases}
x_{\min}+b_b-p_x, & p_x<x_{\min}+b_b,\\
x_{\max}-b_b-p_x, & p_x>x_{\max}-b_b,\\
0, & \mathrm{otherwise},
\end{cases}\\
f_{\mathrm{bnd},y}(\mathbf p)&\ \text{is defined analogously},
\end{aligned}
\label{eq:boundary-pressure}
\end{equation}
其中 $b_b=6$。该分量只提高水平边界裕度，硬边界仍由投影保证。

\paragraph{转弯分量}
若连续控制点形成的局部转角 $\vartheta_k$ 接近或超过阈值，转弯分量施加局部平滑方向。$\vartheta_k$ 按式 \eqref{eq:turning-angle} 计算，但以控制点 $\mathbf q_k$ 替代采样点 $\mathbf p_m$：
\begin{equation}
    \mathbf{F}_{k,\mathrm{curv}}=
    \begin{cases}
    \gamma_{\mathrm{curv}}
    \left[\tfrac{1}{2}(\mathbf{q}_{k-1}+\mathbf{q}_{k+1})-\mathbf{q}_k\right],
    & \vartheta_k\geq \theta_c,\\[4pt]
    \mathbf 0,
    & \mathrm{otherwise},
    \end{cases}
    \label{eq:curvature-field}
\end{equation}
其中 $\gamma_{\mathrm{curv}}=0.22$，$\theta_c=\alpha_{\mathrm{curv}}\theta_{\max}$，$\alpha_{\mathrm{curv}}=0.85$。该分量降低控制多边形局部尖锐程度，并继续受状态权重、逐维截断和范数截断约束。

将全部内部控制点增量向量化后得到未截断 CVF 步长：
\begin{equation}
    \widetilde{\mathbf D}_{\mathrm{CVF}}=
    \mathrm{vec}
    \left(
    [\Delta\mathbf q_1^\mathsf{T},\ldots,\Delta\mathbf q_K^\mathsf{T}]^\mathsf{T}
    \right).
    \label{eq:cvf-vectorization}
\end{equation}
所有分量叠加后进行逐维截断、强度缩放和整体范数截断：
\begin{equation}
    \bar D_{\mathrm{CVF},j}=
    \mathrm{clip}
    \left(
    \widetilde D_{\mathrm{CVF},j},
    -\rho_{\mathrm{step}}(u_j-l_j),
    \rho_{\mathrm{step}}(u_j-l_j)
    \right),
    \label{eq:component-clip}
\end{equation}
\begin{equation}
    \mathbf D_{\mathrm{CVF}}^{(0)}=
    \kappa_{\mathrm{cvf}}\bar{\mathbf D}_{\mathrm{CVF}},
    \label{eq:cvf-strength}
\end{equation}
\begin{equation}
    h_{\mathrm{cvf}}=\rho_{\mathrm{norm}}\|\mathbf u-\mathbf l\|_2,\qquad
    r_{\mathrm{cvf}}=\frac{h_{\mathrm{cvf}}}
    {\|\mathbf D_{\mathrm{CVF}}^{(0)}\|_2+\epsilon_l}.
\end{equation}
\begin{equation}
    \mathbf D_{\mathrm{CVF}}=
    \begin{cases}
    \mathbf D_{\mathrm{CVF}}^{(0)}, &
    \|\mathbf D_{\mathrm{CVF}}^{(0)}\|_2\leq h_{\mathrm{cvf}},\\
    r_{\mathrm{cvf}}\mathbf D_{\mathrm{CVF}}^{(0)}, &
    \mathrm{otherwise}.
    \end{cases}
    \label{eq:norm-clip}
\end{equation}
其中 $\kappa_{\mathrm{cvf}}$ 为 CVF 强度系数。$\rho_{\mathrm{step}}$ 和 $\rho_{\mathrm{norm}}$ 为小比例截断系数，使 CVF 只提供受控方向而不主导单次搜索步长。本文中 $\mathrm{clip}(\cdot,\rho)$ 表示逐维对称截断；若上下界不同，则显式写出上下界。

质量引导方向 $\mathbf D_{\mathrm Q}$ 由当前最优解、采样精英个体和控制点平滑项共同构成：
\begin{equation}
    \mathbf D_{\mathrm Q}=
    \mathrm{clip}
    \left(
    0.6(\mathbf X^*-\mathbf X_i)
    +0.4(\mathbf X_{\mathrm{elite}}-\mathbf X_i)
    +\mathbf D_{\mathrm{sm}},
    \rho_{\mathrm Q}
    \right),
    \label{eq:quality-step}
\end{equation}
其中 $\mathbf D_{\mathrm{sm}}$ 为控制点局部平滑增量，$\rho_{\mathrm Q}$ 表示逐维质量步长上界。

\subsection{稀疏触发与候选接收}

CVF 候选由个体排名、近可行程度、当前状态和代内评价预算共同触发。设第 $t$ 代状态为 $s_t$，对应触发比例为 $r_{s_t}$，则
\begin{equation}
    N_{\mathrm{cvf}}^{(t)}\leq
    \left\lceil r_{s_t}N\right\rceil .
    \label{eq:cvf-quota}
\end{equation}
形成状态优先处理低违反量个体，细化状态降低触发频率以保护可行精英，恢复状态允许少量额外候选恢复搜索活性。

每个被触发的个体先评价基础候选 $\mathbf{X}_{\mathrm{AE}}$。若生成了 $\mathbf{X}_{\mathrm{CVF}}$，则对二者进行局部候选比较。CVF 候选被采用需同时满足两个条件：
\begin{equation}
    \mathbf{X}_{\mathrm{CVF}}
    \prec_{\mathrm{Deb}}
    \mathbf{X}_{\mathrm{AE}},
    \qquad
    J(\mathbf{X}_{\mathrm{CVF}})
    \leq
    J(\mathbf{X}_{\mathrm{AE}}),
    \label{eq:local-cvf-accept}
\end{equation}
其中 $\prec_{\mathrm{Deb}}$ 为第 3 节的可行性优先关系。该保守规则要求 CVF 候选既 Deb 优于基础候选，又不增加综合目标值。因此，CVF 分支并不保证所有生成的 CVF 候选均被用于可行性推进；它只允许相对于基础候选同时满足可行性优先和目标不劣的保守改进进入后续竞争。最终候选再与父代按同一规则比较。

稀疏可行性保持仅对少量候选作碰撞、禁飞区、转弯和风险校正，发生在候选生成之后，作为独立后处理。

\subsection{算法复杂度}

设种群规模为 $N$，最大迭代次数为 $T$，完整路径评价采样点数为 $M$，CVF 低密度采样点数为 $\widetilde M$，内部控制点数为 $K$，空间约束元素数为 $G$。基础搜索总评价规模为 $O(TN)$；单次路径评价生成 B 样条采样点并检查目标与约束，主要代价为
\begin{equation}
    O\big(M(G+K)\big).
\end{equation}

若每代触发个体数为 $N_{\mathrm{cvf}}$，则 CVF 额外候选评价次数为 $O(TN_{\mathrm{cvf}})$，对应完整路径评价代价为 $O(TN_{\mathrm{cvf}}M(G+K))$；低密度场构造开销约为
\begin{equation}
    O\big(TN_{\mathrm{cvf}}\widetilde M(G+K)\big).
\end{equation}
由于稀疏触发和后期节流通常使 $N_{\mathrm{cvf}}\ll N$，额外开销主要由少量关键个体承担，而非全种群强制场更新。

\begin{algorithm}[htbp]
\caption{CVF-AE 的完整算法流程}
\label{alg:cvf-ae}
\scriptsize
\begin{algorithmic}[1]
\Require 地图 $\mathcal{M}$，起点 $\mathbf{s}$，终点 $\mathbf{g}$，种群规模 $N$，最大迭代次数 $T$
\Ensure 最优可行路径 $\mathcal{P}^*$
\State 初始化种群 $\{\mathbf{X}_i\}_{i=1}^{N}$，并将控制点限制在边界内
\State 评价所有个体的 $J(\mathbf{X}_i)$ 与 $V(\mathbf{X}_i)$
\State 按 Deb 可行性优先规则记录当前最优解 $\mathbf{X}^*$
\For{$t=1$ to $T$}
    \State 识别观测状态 $s_{\mathrm{obs}}$，并通过保守调度得到生效状态 $s_t$
    \For{$i=1$ to $N$}
        \State 由基础自适应进化算子生成 $\mathbf{X}_{\mathrm{AE}}$
        \State 评价 $\mathbf{X}_{\mathrm{AE}}$ 的目标值和违反量
        \If{个体 $i$ 满足 CVF 稀疏触发条件}
            \State 构造 $\mathbf{D}_{\mathrm{CVF}}$ 与 $\mathbf{D}_{\mathrm{Q}}$
            \State 按式 \eqref{eq:cvf-candidate} 生成 $\mathbf{X}_{\mathrm{CVF}}$
            \State 评价 $\mathbf{X}_{\mathrm{CVF}}$ 的目标值和违反量
            \If{$\mathbf{X}_{\mathrm{CVF}} \prec_{\mathrm{Deb}} \mathbf{X}_{\mathrm{AE}}$ 且 $J(\mathbf{X}_{\mathrm{CVF}})\leq J(\mathbf{X}_{\mathrm{AE}})$}
                \State $\mathbf{Y}\gets \mathbf{X}_{\mathrm{CVF}}$
            \Else
                \State $\mathbf{Y}\gets \mathbf{X}_{\mathrm{AE}}$
            \EndIf
        \Else
            \State $\mathbf{Y}\gets \mathbf{X}_{\mathrm{AE}}$
        \EndIf
        \If{$\mathbf{Y}$ 被选入稀疏可行性保持集合}
            \State 对 $\mathbf{Y}$ 进行局部路径修复并重新评价
        \EndIf
        \If{$\mathbf{Y} \prec_{\mathrm{Deb}} \mathbf{X}_i$}
            \State $\mathbf{X}_i\gets \mathbf{Y}$
        \EndIf
        \State 更新当前最优解 $\mathbf{X}^*$
    \EndFor
\EndFor
\State \Return 由 $\mathbf{X}^*$ 解码得到的路径 $\mathcal{P}^*$
\end{algorithmic}
\end{algorithm}
